export const sleep = (ms: number) => new Promise((resolve) => setTimeout(resolve, ms));

let lastCallTime = 0;
export const rateLimit = async () => {
  const now = Date.now();
  const elapsed = now - lastCallTime;
  if (elapsed < 3000) {
    await sleep(3000 - elapsed);
  }
  lastCallTime = Date.now();
};

export interface StyleDescriptor {
  /** Injected verbatim into the generation prompt */
  prompt: string;
  /** Blend weight when multiple styles are selected (0.0–1.0) */

  temperatureDelta: number;
}

function mapImageSizeToApi(size: ImageSize): string {
  switch (size) {
    case "720p":   return "0.5K";
    case "1080p":  return "1K";
    case "1440p":  return "2K";
    case "2K":     return "2K";
    case "4K":     return "4K";
    case "8K":     return "4K";
    default:       return "1K";
  }
}

function resolveTemperature(
  model: ModelVersion,
  styles: SchematicStyle[],
  quality: OutputQuality
): number {
  const modelParams = MODEL_DEFAULTS[model] || MODEL_DEFAULTS['gemini-3.1-flash-image'];
  const base = modelParams.temperature;
  const styleDelta = styles.reduce(
    (acc, s) => acc + (STYLE_REGISTRY[s]?.temperatureDelta ?? 0),
    0
  ) / Math.max(styles.length, 1);
  const qualityDelta = quality === "maximum" ? -0.10 : quality === "high" ? -0.05 : 0;

  return Math.max(0.01, Math.min(1.0, base + styleDelta + qualityDelta));
}

function classifyError(raw: unknown, attempt: number): AppError {
  let msg: string;
  if (raw instanceof Error) {
    msg = raw.message;
  } else if (typeof raw === "string") {
    msg = raw;
  } else if (typeof raw === "object" && raw !== null) {
    try {
      msg = JSON.stringify(raw);
    } catch {
      msg = String(raw);
    }
  } else {
    msg = String(raw);
  }
  
  const lc = msg.toLowerCase();
  
  console.log(`[DEBUG] classifyError: Error message: ${msg}`);

  const isOverloaded = 
    lc.includes("rate") || 
    lc.includes("429") || 
    lc.includes("503") || 
    lc.includes("unavailable") || 
    lc.includes("high demand") || 
    lc.includes("overloaded") ||
    lc.includes("exhausted") ||
    lc.includes("timeout") ||
    lc.includes("deadline");

  const retryable = isOverloaded || !(
    lc.includes("api_key") ||
    lc.includes("authentication") ||
    lc.includes("billing") ||
    lc.includes("permission") ||
    lc.includes("invalid argument")
  );
  
  console.log(`[DEBUG] classifyError: Retryable: ${retryable} | IsOverloaded: ${isOverloaded}`);

  let code = ErrorCode.UNKNOWN;
  if (lc.includes("api_key") || lc.includes("api key")) code = ErrorCode.MISSING_API_KEY;
  else if (lc.includes("quota") || lc.includes("exhausted")) code = ErrorCode.QUOTA_EXCEEDED;
  else if (lc.includes("billing"))                        code = ErrorCode.BILLING_REQUIRED;
  else if (isOverloaded)                                  code = ErrorCode.RATE_LIMITED;
  else if (lc.includes("timeout"))                        code = ErrorCode.TIMEOUT;
  else if (lc.includes("network") || lc.includes("fetch"))code = ErrorCode.NETWORK_ERROR;
  else if (lc.includes("no candidates"))                  code = ErrorCode.NO_CANDIDATES;
  else if (lc.includes("text instead of image"))          code = ErrorCode.TEXT_RESPONSE;
  else if (lc.includes("no image"))                       code = ErrorCode.NO_IMAGE_IN_RESPONSE;
  else if (lc.includes("content") && lc.includes("filter"))code = ErrorCode.CONTENT_FILTERED;

  return { code, message: msg, retryable, attempt, originalError: raw };
}

const DEFAULT_RETRY: RetryPolicy = {
  maxAttempts: 5,
  baseDelayMs: 1500,
  maxDelayMs: 8000,
  backoffMultiplier: 2.0,
};

export async function generateContentStreamAndLog(
  ai: any,
  caller: string,
  model: string,
  params: any
): Promise<{ text: string }> {
  const startTime = Date.now();
  
  // Transform deprecated / high-demand preview model to the stable 3.5-flash model
  const resolvedModel = model === "gemini-3-flash-preview" ? "gemini-3.5-flash" : model;
  
  console.info(`[gemini] Starting generation with model: ${resolvedModel} (requested: ${model}) - Caller: ${caller}`);

  try {
    const msgId = Math.random().toString(36).substring(2, 9);
    
    const stream = await ai.models.generateContentStream({ ...params, model: resolvedModel });
    let text = "";
    let chunkCount = 0;
    let lastChunkTime = Date.now();

    for await (const chunk of stream) {
      if (chunk.text) {
        text += chunk.text;
        chunkCount++;
        const now = Date.now();
        const timeSinceLast = now - lastChunkTime;
        lastChunkTime = now;
        
        // Stream to browser console which is captured by the AI Studio terminal
        console.log(`[${caller}] Chunk #${chunkCount} | +${timeSinceLast}ms | Text Length: ${text.length} | Preview: ${chunk.text.substring(0, 100).replace(/\n/g, '\\n')}...`);
      }
    }
    
    const duration = Date.now() - startTime;
    console.info(`[gemini] ✓ Completed generation from ${caller} in ${duration}ms. Total chunks: ${chunkCount}, Total length: ${text.length}`);

    return { text };
  } catch (error) {
    const duration = Date.now() - startTime;
    console.error(`[gemini] ❌ Error during generation from ${caller} after ${duration}ms:`, error);
    
    throw error;
  }
}

export async function withRetry<T>(
  fn: (attempt: number, model: string) => Promise<T>,
  modelType: 'text' | 'image' = 'text',
  policy: RetryPolicy = DEFAULT_RETRY,
  primaryModel?: string
): Promise<T> {
  let lastError: AppError | null = null;

  for (let attempt = 1; attempt <= policy.maxAttempts; attempt++) {
    const model = getModelForAttempt(attempt, modelType, primaryModel);
    try {
      return await fn(attempt, model);
    } catch (raw) {
      lastError = classifyError(raw, attempt);

      if (!lastError.retryable) throw lastError;

      if (attempt < policy.maxAttempts) {
        const jitter = Math.random() * 500;
        const delay = Math.min(
          policy.baseDelayMs * Math.pow(policy.backoffMultiplier, attempt - 1) + jitter,
          policy.maxDelayMs
        );
        const nextModel = getModelForAttempt(attempt + 1, modelType, primaryModel);
        const errorPreview = lastError.message.substring(0, 100).replace(/\n/g, ' ');
        
        console.warn(
          `[gemini] Attempt ${attempt}/${policy.maxAttempts} failed — ${lastError.code} (${errorPreview}). Switch to ${nextModel}. Retrying in ${Math.round(delay)}ms.`
        );
        await new Promise<void>((r) => setTimeout(r, delay));
      }
    }
  }
  throw lastError;
}

export function getModelForAttempt(attempt: number, type: 'text' | 'image' = 'text', primaryModel?: string): string {
  let models: string[] = [];

  if (type === 'text') {
    models = [
      "gemini-3.5-flash",
      "gemini-3.1-flash-lite",
      "gemini-3-flash-preview",
      "gemini-2.5-flash"
    ];
  } else {
    models = [
      "gemini-3.1-flash-image",
      "gemini-3.5-flash",
      "gemini-2.5-flash-image"
    ];
  }

  // If a primaryModel is strictly requested via the UI, ensure it's the first one attempted
  // and we rotate through the others as fallbacks.
  if (primaryModel && models.includes(primaryModel)) {
    const primaryIndex = models.indexOf(primaryModel);
    models.splice(primaryIndex, 1);
    models.unshift(primaryModel);
  }

  const index = (attempt - 1) % models.length;
  return models[index];
}

function validateBase64Image(data: string, caller: string): void {
  if (!data || typeof data !== "string") {
    throw {
      code: ErrorCode.INVALID_IMAGE,
      message: `[${caller}] base64Image is null or not a string.`,
      retryable: false,
    } satisfies AppError;
  }
  if (data.length < 100) {
    throw {
      code: ErrorCode.INVALID_IMAGE,
      message: `[${caller}] base64Image appears truncated (length: ${data.length}).`,
      retryable: false,
    } satisfies AppError;
  }
}

function validateApiKey(caller: string): void {
  if (!process.env.GEMINI_API_KEY) {
    throw {
      code: ErrorCode.MISSING_API_KEY,
      message: `[${caller}] GEMINI_API_KEY is not set. Call window.aistudio.openSelectKey() to configure.`,
      retryable: false,
    } satisfies AppError;
  }
}

function extractImageFromResponse(candidate: {
  content?: { parts?: Array<{ inlineData?: { data?: string; mimeType?: string }; text?: string }> };
  finishReason?: string;
}): { data: string; mimeType: ImageMimeType } {
  const parts = candidate.content?.parts ?? [];

  for (const part of parts) {
    if (part.inlineData?.data) {
      return {
        data: part.inlineData.data,
        mimeType: (part.inlineData.mimeType as ImageMimeType) ?? "image/png",
      };
    }
  }

  const textPart = parts.find((p) => p.text);
  if (textPart?.text) {
    const preview = textPart.text.substring(0, 400);
    throw {
      code: ErrorCode.TEXT_RESPONSE,
      message: `Model returned a text response instead of an image: "${preview}…"`,
      retryable: true,
    } satisfies AppError;
  }

  throw {
    code: ErrorCode.NO_IMAGE_IN_RESPONSE,
    message: `No image data in response. Finish reason: ${candidate.finishReason ?? "UNKNOWN"}`,
    retryable: true,
  } satisfies AppError;
}

/**
 * Automatically computes the pixel-perfect aspect ratio of the input image on the client-side.
 * This is 100% deterministic, 100% reliable, zero-latency, and zero-cost compared to an LLM call.
 */
export function getPixelAspectRatio(base64Data: string, mimeType: string): Promise<AspectRatio> {
  if (typeof window === 'undefined' || typeof Image === 'undefined') {
    return Promise.resolve("4:3"); // Safe landscape default if in non-browser env
  }
  return new Promise((resolve) => {
    try {
      const img = new Image();
      img.onload = () => {
        const width = img.naturalWidth || img.width;
        const height = img.naturalHeight || img.height;
        if (!width || !height) {
          resolve("4:3");
          return;
        }
        const ratio = width / height;
        
        // Supported standard output ratios
        const candidates: { ratio: AspectRatio; value: number }[] = [
          { ratio: "1:1", value: 1.0 },
          { ratio: "4:3", value: 4 / 3 },     // 1.333
          { ratio: "3:4", value: 3 / 4 },     // 0.75
          { ratio: "16:9", value: 16 / 9 },   // 1.777
          { ratio: "9:16", value: 9 / 16 },   // 0.5625
          { ratio: "1:4", value: 0.25 },
          { ratio: "1:8", value: 0.125 },
          { ratio: "4:1", value: 4.0 },
          { ratio: "8:1", value: 8.0 },
        ];
        
        let bestRatio: AspectRatio = "4:3";
        let minDiff = Infinity;
        
        for (const candidate of candidates) {
          const diff = Math.abs(ratio - candidate.value);
          if (diff < minDiff) {
            minDiff = diff;
            bestRatio = candidate.ratio;
          }
        }
        
        console.info(`[gemini] Programmatic aspect ratio auto-detected: ${bestRatio} (original: ${width}x${height}, calculated ratio: ${ratio.toFixed(4)})`);
        resolve(bestRatio);
      };
      
      img.onerror = () => {
        console.warn("[gemini] Error loading image for aspect ratio evaluation, defaulting to 4:3");
        resolve("4:3");
      };
      
      if (base64Data.startsWith('data:')) {
        img.src = base64Data;
      } else {
        const cleanMime = mimeType || 'image/png';
        img.src = `data:${cleanMime};base64,${base64Data}`;
      }
    } catch (e) {
      console.error("[gemini] Error computing aspect ratio:", e);
      resolve("4:3");
    }
  });
}

/**
 * Uses programmatic client-side analyzer to determine the exact aspect ratio of the input image.
 * This guarantees consistent output image dimensions without wasteful margins or stretching.
 */
export async function detectOptimalAspectRatio(
  base64Image: string,
  mimeType: string,
  model: ModelVersion,
  styles?: string[]
): Promise<AspectRatio> {
  const caller = "detectOptimalAspectRatio";

  if (styles?.includes("mobile_responsive")) {
    return "9:16";
  }

  try {
    return await getPixelAspectRatio(base64Image, mimeType);
  } catch (e) {
    console.warn(`[gemini] Programmatic aspect ratio detection failed, falling back to 4:3`, e);
    return "4:3";
  }
}

export type SchematicType =
  | "exploded_view"
  | "cross_section"
  | "wiring_diagram"
  | "assembly_drawing"
  | "parts_breakdown"
  | "unknown";

interface ClassificationResult {
  type: SchematicType;
  confidence: number;
  dominantFeatures: string[];
  estimatedPartCount: number;
  hasLeaderLines: boolean;
  hasCalloutBubbles: boolean;
  viewAngle: "isometric" | "orthographic" | "perspective" | "flat";
}

const CLASSIFICATION_PROMPT = `You are a precision optical classifier for engineering drawings.
Analyze this image and return a JSON object ONLY. No explanation.
Schema:
{
  "type": "exploded_view" | "cross_section" | "wiring_diagram" | "assembly_drawing" | "parts_breakdown" | "unknown",
  "confidence": 0.0-1.0,
  "dominantFeatures": ["string"],  // max 5, e.g. ["leader_lines", "hatching", "coil_springs"]
  "estimatedPartCount": number,    // integer 0-999
  "hasLeaderLines": boolean,
  "hasCalloutBubbles": boolean,
  "viewAngle": "isometric" | "orthographic" | "perspective" | "flat"
}`;

const classificationCache = new Map<string, ClassificationResult>();

export async function classifySchematic(
  base64Image: string,
  mimeType: string
): Promise<ClassificationResult> {
  const cacheKey = base64Image.substring(0, 80);
  if (classificationCache.has(cacheKey)) {
    return classificationCache.get(cacheKey)!;
  }

  const fallback: ClassificationResult = {
    type: "unknown",
    confidence: 0,
    dominantFeatures: [],
    estimatedPartCount: 0,
    hasLeaderLines: false,
    hasCalloutBubbles: false,
    viewAngle: "flat",
  };

  try {
    const result = await withRetry(async (attempt, model) => {
      const ai = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY });
      const response = await ai.models.generateContent({
        model,
        contents: {
          parts: [
            { text: CLASSIFICATION_PROMPT },
            { inlineData: { mimeType, data: base64Image } },
          ],
        },
        config: { temperature: 0.05, responseMimeType: "application/json" },
      });
      const clean = (response.text || "{}").replace(/```json|```/g, "").trim();
      return JSON.parse(clean) as ClassificationResult;
    }, 'text');

    classificationCache.set(cacheKey, result);
    return result;
  } catch (e) {
    console.warn("[gemini] classifySchematic failed, using fallback:", e);
    return fallback;
  }
}

function buildSpecializationBlock(classification: ClassificationResult): string {
  const base = SPECIALIZATION_BLOCKS[classification.type];

  // Append dynamic feature warnings based on detected content
  const warnings: string[] = [];
  if (classification.estimatedPartCount > 20) {
    warnings.push(
      `⚠ HIGH PART COUNT: ~${classification.estimatedPartCount} discrete parts detected. You MUST render every single one. Do NOT group, merge, or simplify any cluster.`
    );
  }
  if (classification.dominantFeatures.includes("coil_springs")) {
    warnings.push(
      `⚠ SPRING DETECTED: Coil springs present. You MUST render correct helix geometry with consistent pitch and uniform coil diameter. Never draw a spring as a zigzag or cylinder.`
    );
  }
  if (classification.dominantFeatures.includes("hatching")) {
    warnings.push(
      `⚠ SECTION HATCHING DETECTED: Render ALL hatched regions with geometrically perfect parallel lines at the same angle and spacing as the original.`
    );
  }
  if (classification.viewAngle === "isometric" && !classification.dominantFeatures.includes("isometric")) {
    warnings.push(
      `⚠ ISOMETRIC VIEW: Maintain true 30° isometric projection. Do not distort to perspective.`
    );
  }

  return warnings.length > 0
    ? `${base}\n\n<DYNAMIC_FEATURE_WARNINGS>\n${warnings.join("\n")}\n</DYNAMIC_FEATURE_WARNINGS>`
    : base;
}

Produce the image now.
</RETRY_CORRECTION>`,
};

function buildAttemptCorrectionBlock(context: GenerationFailureContext): string {
  if (context.attempt <= 1) return "";

  const blocks: string[] = [];

  if (context.errorCode && FAILURE_CORRECTION_DIRECTIVES[context.errorCode]) {
    blocks.push(FAILURE_CORRECTION_DIRECTIVES[context.errorCode]!);
  }

  // Generic escalation for later attempts
  if (context.attempt >= 3) {
    blocks.push(`
<RETRY_ESCALATION — Attempt ${context.attempt}>
This is attempt ${context.attempt}. Previous attempts have failed. Apply MAXIMUM precision:
• Double-check every geometric constraint before finalizing
• Verify part count against the audit report
• Confirm background is #FFFFFF with zero shadows
• Output the image with zero accompanying text
</RETRY_ESCALATION>`);
  }

  return blocks.join("\n\n");
}

// STYLE CONFLICT RESOLVER
// ============================================================================

type StyleConflictRule = {
  styles: [SchematicStyle, SchematicStyle];
  resolution: "keep_first" | "keep_second" | "merge_note";
  mergeNote?: string;
};

const STYLE_CONFLICTS: StyleConflictRule[] = [
  {
    styles: ["minimalist", "realistic"],
    resolution: "merge_note",
    mergeNote: `STYLE CONFLICT RESOLVED — Minimalist + Realistic: Apply minimalist line geometry (clean strokes, no fills) with realistic material surface hints on edges only. No fills, no ambient light — use line weight variation to imply material.`,
  },
  {
    styles: ["blueprint", "modern_realism"],
    resolution: "merge_note",
    mergeNote: `STYLE CONFLICT RESOLVED — Blueprint + Modern Realism: Use blueprint colour palette (deep blue bg, white lines) but apply realistic material edge highlighting. No photorealistic fills.`,
  },
  {
    styles: ["minimalist", "artistic"],
    resolution: "keep_first",
  },
  {
    styles: ["patent", "realistic"],
    resolution: "keep_first", // Patent rules forbid gradients/colour — enforce them
  },
  {
    styles: ["patent", "isometric"],
    resolution: "merge_note",
    mergeNote: `STYLE CONFLICT RESOLVED — Patent + Isometric: Use isometric 30° projection geometry, but render in strict patent style (black ink only, hatching for shading, no colour fills).`,
  },
];

export function resolveStyleConflicts(styles: SchematicStyle[]): {
  resolvedStyles: SchematicStyle[];
  conflictNote: string;
} {
  let resolvedStyles = [...styles];
  const notes: string[] = [];

  for (const rule of STYLE_CONFLICTS) {
    const [a, b] = rule.styles;
    if (resolvedStyles.includes(a) && resolvedStyles.includes(b)) {
      if (rule.resolution === "keep_first") {
        resolvedStyles = resolvedStyles.filter((s) => s !== b);
        notes.push(`Removed conflicting style '${b}' in favour of '${a}'.`);
      } else if (rule.resolution === "keep_second") {
        resolvedStyles = resolvedStyles.filter((s) => s !== a);
        notes.push(`Removed conflicting style '${a}' in favour of '${b}'.`);
      } else if (rule.resolution === "merge_note" && rule.mergeNote) {
        notes.push(rule.mergeNote);
      }
    }
  }

  return {
    resolvedStyles,
    conflictNote: notes.join("\n"),
  };
}

export async function enhanceSchematic(
  base64Image: string,
  mimeType: string,
  styles: SchematicStyle[] = ["modern"],
  keepLabels: boolean = true,
  aspectRatio: AspectRatioOption = "1:1",
  imageSize: ImageSize = "1440p",
  model: ModelVersion = "gemini-3.1-flash-image",
  customPrompt: string = "",
  preserveGeometry: boolean = true,
  enhanceDetails: boolean = true,
  outputQuality: OutputQuality = "high",
  referenceImages?: { url: string; mimeType: string }[],
  seed?: number,
  mode: 'enhance' | 'creative' | 'style_transfer' | 'upscale' = 'enhance'
): Promise<{ imageUrl: string; aspectRatio: AspectRatio }> {
  const caller = "enhanceSchematic";

  // Validate
  validateApiKey(caller);
  validateBase64Image(base64Image, caller);
  if (referenceImages && referenceImages.length > 0) {
    referenceImages.forEach(ref => validateBase64Image(ref.url, caller));
  }

  // Normalise styles and resolve conflicts
  const normalizedStyles = normalizeStyles(styles);
  const { resolvedStyles, conflictNote } = resolveStyleConflicts(normalizedStyles);

  // Handle Auto Aspect Ratio
  let targetAspectRatio: AspectRatio;
  if (aspectRatio === "auto") {
    try {
      targetAspectRatio = await detectOptimalAspectRatio(base64Image, mimeType, model, resolvedStyles);
    } catch (e) {
      console.error("[gemini] Failed to detect aspect ratio, defaulting to 1:1", e);
      targetAspectRatio = "1:1";
    }
  } else {
    targetAspectRatio = aspectRatio;
  }

  const config: SchematicEnhancementConfig = {
    mode,
    styles: resolvedStyles,
    keepLabels,
    aspectRatio: targetAspectRatio,
    imageSize,
    model,
    customPrompt,
    preserveGeometry,
    enhanceDetails,
    outputQuality,
    seed,
  };

  // Build prompt components
  const styleBlock   = buildStyleBlock(resolvedStyles);
  const labelDirective = buildLabelDirective(keepLabels);
  const hasReferenceImages = referenceImages !== undefined && referenceImages.length > 0;
  let prompt       = buildEnhancePrompt(config, styleBlock, labelDirective, hasReferenceImages);

  if (conflictNote) {
    prompt += `\n\n<STYLE_RESOLUTION_NOTES>\n${conflictNote}\n</STYLE_RESOLUTION_NOTES>`;
  }

  console.info(`[gemini] ${caller} — Received entry request. Converted base64 length: ${base64Image.length} bytes | mimeType: ${mimeType} | styles: ${resolvedStyles.join(", ")} | keepLabels: ${keepLabels} | aspectRatioOption: ${aspectRatio} | imageSize: ${imageSize} | model: ${model}`);

  const promises: Promise<void>[] = [];

  // Enable text-based pre-processing audits for absolute highest fidelity and colorized rendering
  console.info(`[gemini] ${caller} — Dispatching schematic classification tasks...`);
  const classificationPromise = classifySchematic(base64Image, mimeType);

  console.info(`[gemini] ${caller} — Dispatching schematic audit tasks...`);
  const auditPromise = classificationPromise.then(async (classification) => {
    console.info(`[gemini] Classification: ${classification.type} (${(classification.confidence * 100).toFixed(0)}% confidence) | ~${classification.estimatedPartCount} parts`);

    if (config.mode !== 'creative') {
      const specializationBlock = buildSpecializationBlock(classification);
      prompt += `\n\n${specializationBlock}`;

      console.info(`[gemini] ${caller} — Successfully compiled specialization block. Appending to prompt...`);
    }

    const auditInstruction = config.mode === 'creative'
      ? `The following is a geometric and label audit performed by an engineer. Use this inventory as a conceptual understanding of the physical parts that make up this mechanical assembly. As you creatively redesign and rethink the rendering, ensure all major structural components exist in your new vision, but you are free to artistically reshape, style, and reinterpret them:\n\n`
      : `The following is a meticulous geometric and label audit performed by a master engineer. You MUST use this inventory to ensure absolute 1-to-1 pixel fidelity and prevent hallucinations. Do not miss any label or part outlined here. Do not mutate shapes (e.g., do not add threads if the audit says "smooth pin"):\n\n`;

    const auditReport = await extractSchematicAudit(base64Image, mimeType, classification);
    console.info(`[gemini] ${caller} — Successfully compiled schematic audit. Appending technical inventory guidelines...`);
    prompt += `\n\n<SCHEMATIC_AUDIT_REPORT>\n${auditInstruction}${auditReport}\n</SCHEMATIC_AUDIT_REPORT>`;
  });
  promises.push(auditPromise);

  if (hasReferenceImages && referenceImages && config.mode !== 'style_transfer') {
    console.info(`[gemini] ${caller} — Dispatching material color description tasks from reference files...`);
    const materialPromise = extractMaterialDescription(base64Image, mimeType, referenceImages).then(materialDescription => {
      console.info(`[gemini] ${caller} — Successfully compiled material mapping description. Appending colorization keys...`);
      prompt += `\n\n<MATERIAL_AND_COLOR_GUIDE>\nThe following is a detailed description of the materials, colors, and finishes extracted from the reference photos of the real product. You MUST use this description to accurately colorize and texture the schematic diagram:\n\n${materialDescription}\n</MATERIAL_AND_COLOR_GUIDE>`;
    });
    promises.push(materialPromise);
  }

  await Promise.all(promises);

  // Resolve temperature
  const modelParams  = MODEL_DEFAULTS[model] || MODEL_DEFAULTS['gemini-3.1-flash-image'];
  const temperature  = resolveTemperature(model, resolvedStyles, outputQuality);

  console.info(`[gemini] ${caller} — Setup completed. Final prompt character size: ${prompt.length}. Invoking retry handler for generation...`);

  let lastAttemptError: unknown;

  return withRetry(async (attempt, model) => {
    // Re-initialise client on every attempt to guarantee fresh API key pickup
    const ai = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY });
    const currentModel = model;

    console.info(
      `[gemini] [INITIATED] ${caller} — attempt ${attempt} | model=${currentModel} | quality=${outputQuality} | styles=${resolvedStyles.join(",")} | ratio=${targetAspectRatio} | temp=${temperature.toFixed(2)}`
    );
    console.info(`[gemini] [ALERT] Sending monolithic technical drawing synthesis request to "${currentModel}". IMAGE COMPILING TENDS TO TAKE 10-25 SECONDS. This operation cannot be streamed. Please stand by...`);

    const correctionBlock = buildAttemptCorrectionBlock({
      attempt,
      errorCode: attempt > 1 ? classifyError(lastAttemptError, attempt - 1).code : undefined,
    });

    const finalPrompt = correctionBlock
      ? `${correctionBlock}\n\n${prompt}`
      : prompt;

    const parts: any[] = [
      { text: finalPrompt },
      { inlineData: { mimeType, data: base64Image } },
    ];

    if (config.mode === 'style_transfer' && referenceImages) {
      referenceImages.forEach(ref => {
        const [, base64Data] = ref.url.split(';base64,');
        parts.push({ inlineData: { mimeType: ref.mimeType, data: base64Data || ref.url } });
      });
    }

    const generationConfig: any = {
      temperature,
      topP: modelParams.topP,
      topK: modelParams.topK,
      seed: seed !== undefined ? seed : 42,
    };

    if (currentModel.includes('image')) {
      generationConfig.imageConfig = { imageSize: mapImageSizeToApi(imageSize) };
      const hasImageInput = parts.some((p: any) => p.inlineData);
      if (!hasImageInput) {
        generationConfig.imageConfig.aspectRatio = targetAspectRatio;
      }
    }

    const startTime = Date.now();
    try {
      const response = await ai.models.generateContent({
        model: currentModel,
        contents: {
          parts,
        },
        config: generationConfig,
      });
      const duration = Date.now() - startTime;
      console.info(`[gemini] [SUCCESS] Received technical diagram content candidate successfully from Gemini "${currentModel}" in ${duration}ms! Unpacking payload...`);

      const candidate = response.candidates?.[0];
      if (!candidate) {
        console.error(`[gemini] [FAILED] "${currentModel}" returned zero candidates. Throwing AppError...`);
        throw {
          code: ErrorCode.NO_CANDIDATES,
          message: "Gemini API returned zero candidates.",
          retryable: true,
        } satisfies AppError;
      }

      // Warn on unusual finish reasons but don't block extraction
      const fr = candidate.finishReason;
      if (fr && !["STOP", "MAX_TOKENS"].includes(fr)) {
        console.warn(`[gemini] Warning finish reason for model execution: ${fr}`);
      }

      const { data, mimeType: outMime } = extractImageFromResponse(candidate as Parameters<typeof extractImageFromResponse>[0]);
      console.info(`[gemini] [FINISHED] Successfully parsed enhanced schematic image stream: mimeType=${outMime}, dataSize=${data.length} characters.`);
      return {
        imageUrl: `data:${outMime};base64,${data}`,
        aspectRatio: targetAspectRatio
      };
    } catch (e) {
      lastAttemptError = e;
      throw e;
    }
  }, 'image', undefined, config.model);
}

const productSpecCache = new Map<string, string>();

export async function extractProductSpecification(
  base64Image: string,
  mimeType: string,
  mode?: 'product_hero' | 'hardware_part' | 'clone'
): Promise<string> {
  const caller = "extractProductSpecification";
  validateApiKey(caller);

  // Use length and a slice from the middle of the base64 string to ensure a more unique cache key. Add v4 to invalidate bad caches.
  const cacheKey = `v4_len_${base64Image.length}_mid_${base64Image.substring(Math.floor(base64Image.length / 2), Math.floor(base64Image.length / 2) + 100)}_${mode}`;
  if (productSpecCache.has(cacheKey)) {
    console.info(`[gemini] ${caller} — returning cached product specification`);
    return productSpecCache.get(cacheKey)!;
  }

  const prompt = `You are a master industrial designer, macro photographer, and mechanical engineer.
Analyze this raw photo or product image and extract an extremely precise, professional, and dense textual specification for recreation purposes.

We need to feed this highly descriptive specification into an AI image generator to produce a highly-accurate, photorealistic image of this exact physical product.

Provide an exhaustive and detailed analysis of the following categories:

1. PRODUCT IDENTITY & INVENTORY:
  - Exact brand name, logos, typography, styling, trademark markings, and sub-labels.
  - Precise list of all items present in the reference image. Count items accurately to prevent duplication or omission! Describe exactly what the object is (e.g. a specific type of screw like a "#4-40 fillister head machine screw", a specific hardware tool, a piece of equipment, or retail packaging).

2. PRODUCT DESIGN & MECHANICAL ANATOMY:
  - Detailed physical structure of each object.
  - For hardware/fasteners (like screws, bolts, nuts), explicitly state the head type (e.g., pan, flat, truss, binder, fillister), drive type (Phillips, slotted, hex), thread density, and tip style.

export async function regenerateImage(
  base64Image: string,
  mimeType: string,
  customPrompt: string = "",
  aspectRatio: AspectRatioOption = "1:1",
  imageSize: ImageSize = "1440p",
  model: ModelVersion = "gemini-3.1-flash-image",
  mode: 'product_hero' | 'hardware_part' | 'clone' = 'product_hero',
  onProgress?: (message: string) => void
): Promise<{ imageUrl: string; aspectRatio: AspectRatio }> {
  const caller = "regenerateImage";

  validateApiKey(caller);
  validateBase64Image(base64Image, caller);
  
  const logProgress = (msg: string) => {
    console.info(`[gemini] ${caller} — ${msg}`);
    if (onProgress) onProgress(msg);
  };

  // Handle Auto Aspect Ratio
  let targetAspectRatio: AspectRatio;
  if (aspectRatio === "auto") {
    try {
      logProgress("Detecting optimal aspect ratio...");
      targetAspectRatio = await detectOptimalAspectRatio(base64Image, mimeType, model);
      logProgress(`Aspect ratio detected: ${targetAspectRatio}`);
    } catch (e) {
      console.error("[gemini] Failed to detect aspect ratio, defaulting to 1:1", e);
      targetAspectRatio = "1:1";
    }
  } else {
    targetAspectRatio = aspectRatio;
  }

  // Define strategy parameters based on selected mode
  let finalPrompt = "";
  let useParts: any[] = [];
  const modelParams = MODEL_DEFAULTS[model] || MODEL_DEFAULTS['gemini-3.1-flash-image'];
  const temperature = mode === 'clone' 
    ? 0.05 
    : mode === 'hardware_part' 
      ? 0.15 
      : 0.75;

  if (mode === 'clone') {
    // Standard clone: pass input photo as reference and keep layout 1:1 with low temperature
    finalPrompt = buildPrompt(mode, customPrompt);
    useParts = [
      { text: finalPrompt },
      { inlineData: { mimeType, data: base64Image } },
    ];
    logProgress("Clone mode active: strict 1:1 visual anchoring configured.");
  } else {
    // Creative or POV Mode: Use Vision-to-Text extraction to form a master catalog prompt
    let productSpec = "";
    try {
      logProgress("Stage 1/2: Executing pre-processing vision audit for premium product specification...");
      productSpec = await extractProductSpecification(base64Image, mimeType, mode);
      logProgress("Stage 1/2: Product specifications extracted successfully. \n\n" + productSpec);
    } catch (e) {
      console.warn(`[gemini] ${caller} — vision pre-processing failed due to API limits or high demand. Falling back to direct multimodal reference image transfer!`, e);
      logProgress("Stage 1/2 WARNING: Vision analysis failed. Falling back to direct image reference.");
    }

    const hasReferenceImage = true;

    if (productSpec) {
      finalPrompt = `${buildPrompt(mode, customPrompt, hasReferenceImage)}

PRODUCT STRUCTURE TO RENDERING SPECIFICATION:
<HARDWARE_AUDIT_REPORT>
The following is a meticulous geometric specification and measurements audit performed by a master engineer. You MUST use this inventory to ensure absolute 1-to-1 pixel fidelity and prevent hallucinations. Do not mutate shapes. If the dimensions or aspect ratios mentioned here differ from your visual assumption of the image, the text below is the absolute truth.

${productSpec}
</HARDWARE_AUDIT_REPORT>
`;
    } else {
      finalPrompt = buildPrompt(mode, customPrompt, hasReferenceImage);
    }

    useParts = [
      { text: finalPrompt },
      { inlineData: { mimeType, data: base64Image } }
    ];
    logProgress("Stage 2/2: Proceeding with multimodal structural synthesis. Anchoring output to physical image reference...");
  }

  const regeneratePolicy: RetryPolicy = {
    maxAttempts: 5,
    baseDelayMs: 3000,
    maxDelayMs: 15000,
    backoffMultiplier: 2.0,
  };

  return withRetry(async (attempt, currentModel) => {
    const ai = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY });

    logProgress(`Generation Attempt ${attempt} | Model = ${currentModel} | Temp = ${temperature.toFixed(2)}`);

    const response = await ai.models.generateContent({
      model: currentModel,
      contents: {
        parts: useParts,
      },
      config: {
        imageConfig: (() => {
          const cfg: any = { imageSize: mapImageSizeToApi(imageSize) };
          if (!useParts.some((p: any) => p.inlineData)) cfg.aspectRatio = targetAspectRatio;
          return cfg;
        })(),
        temperature,
        topP: modelParams.topP,
        topK: modelParams.topK,
      },
    });

    const candidate = response.candidates?.[0];
    if (!candidate) {
      throw {
        code: ErrorCode.NO_CANDIDATES,
        message: "Gemini API returned zero candidates.",
        retryable: true,
      } satisfies AppError;
    }

    logProgress(`Stage 2/2 SUCCESS: Received candidate natively generated from structural text specification.`);

    const { data, mimeType: outMime } = extractImageFromResponse(candidate as Parameters<typeof extractImageFromResponse>[0]);
    return {
      imageUrl: `data:${outMime};base64,${data}`,
      aspectRatio: targetAspectRatio,
    };
  }, 'image', regeneratePolicy, model);
}

/**
 * Detects if a label contains multiple distinct part numbers (e.g., list of variants or equations mapping parts)
 * and normalizes it to "Details" to consolidate hotspots for multi-part stacks.
 */
export function normalizeLabelToDetails(label: string): string {
  if (!label) return label;
  return label.trim();
}

/** Normalise styles input to a valid, deduplicated array */
export function normalizeStyles(
  input: SchematicStyle[] | SchematicStyle | undefined | null
): SchematicStyle[] {
  if (!input) return ["modern"];
  const arr = Array.isArray(input) ? input : [input];
  const valid = arr.filter((s) => s in STYLE_REGISTRY) as SchematicStyle[];
  return valid.length > 0 ? [...new Set(valid)] : ["modern"];
}

/** Human-readable style label */
export function getStyleLabel(style: SchematicStyle): string {
  return STYLE_REGISTRY[style]?.label ?? style;
}

/** Human-readable style description for tooltips */
export function getStyleDescription(style: SchematicStyle): string {
  const prompt = STYLE_REGISTRY[style]?.prompt ?? "";
  // Extract first bullet as description
  const match = prompt.match(/•\s+(.+)/);
  return match ? match[1].trim() : style;
}

/** All available styles */
export function getAvailableStyles(): SchematicStyle[] {
  return Object.keys(STYLE_REGISTRY) as SchematicStyle[];
}

/** Validate a configuration before sending to the API */
export function validateConfig(
  config: Partial<SchematicEnhancementConfig>
): ValidationResult {
  const errors: string[] = [];
  const warnings: string[] = [];

  if (!config.styles || config.styles.length === 0) {
    warnings.push("No styles specified — defaulting to 'modern'.");
  } else {
    const invalid = config.styles.filter((s) => !(s in STYLE_REGISTRY));
    if (invalid.length > 0) {
      errors.push(`Unknown styles: ${invalid.join(", ")}`);
    }
    if (config.styles.length > 3) {
      warnings.push(
        "More than 3 styles may produce inconsistent output. Recommend 1–3."
      );
    }
  }

  if (config.customPrompt && config.customPrompt.length > 1000) {
    warnings.push(
      "Custom prompt exceeds 1000 characters; extremely long instructions may be partially ignored."
    );
  }

  return {
    valid: errors.length === 0,
    errors,
    warnings,
  };
}

/** Estimate processing complexity for a given configuration */
export function estimateComplexity(
  config: Partial<SchematicEnhancementConfig>
): ProcessingComplexity {
  let score = 0;
  score += (config.styles?.length ?? 1) * 2;
  score += config.outputQuality === "high" ? 3 : config.outputQuality === "maximum" ? 6 : 1;
  score += config.customPrompt && config.customPrompt.trim().length > 0 ? 2 : 0;
  score += config.enhanceDetails ? 2 : 0;
  score += config.preserveGeometry ? 1 : 0;

  if (score <= 6)  return "low";
  if (score <= 12) return "medium";
  return "high";
}

/** Estimated latency in seconds (rough heuristic for UI hints) */
export function estimateLatencySeconds(
  model: ModelVersion,
  imageSize: ImageSize,
  quality: OutputQuality
): number {
  const base: Record<ModelVersion, number> = {
    "gemini-3.1-flash-image": 8,
    "gemini-3.5-flash": 4,
    "gemini-2.5-flash-image": 8,
    "gemini-3.1-flash-lite": 2,
    "gemini-3-flash-preview": 4,
    "gemini-2.5-flash": 3,
  };
  const sizeMultiplier: Record<ImageSize, number> = {
    "720p": 0.5, "1080p": 1.0, "1440p": 1.5, "2K": 1.8, "4K": 4.0, "8K": 8.0,
  };
  const qualityMultiplier: Record<OutputQuality, number> = {
    standard: 1.0, high: 1.3, maximum: 1.7,
  };

  return Math.round(
    base[model] * sizeMultiplier[imageSize] * qualityMultiplier[quality]
  );
}
