import React, { useState, useRef, useEffect } from 'react';
import JSZip from 'jszip';
import { saveAs } from 'file-saver';
import { Upload, X, Loader2, Sparkles, Image as ImageIcon, Download, FolderOpen, RefreshCw, AlertCircle, CheckCircle2, Save, CloudLightning, Trash2, Play, Pause, Square, ChevronDown, ChevronUp } from 'lucide-react';
import { regenerateImage, refineImage } from '../lib/gemini';
import { AspectRatioOption, ImageSize, ModelVersion, ASPECT_RATIO_LABELS, MODEL_LABELS, IMAGE_SIZE_LABELS } from '../types';
import localforage from 'localforage';

interface RegenerationItem {
  id: string;
  filename: string;
  originalDataUri: string;
  generatedDataUri?: string;
  status: 'pending' | 'processing' | 'success' | 'error';
  error?: string;
}

interface ImageRegeneratorProps {
  onClose: () => void;
}

export function ImageRegenerator({ onClose }: ImageRegeneratorProps) {
  const [items, setItems] = useState<RegenerationItem[]>([]);
  const [itemsLoaded, setItemsLoaded] = useState(false);
  const [selectedItem, setSelectedItem] = useState<RegenerationItem | null>(null);
  const [showClearConfirm, setShowClearConfirm] = useState(false);
  const [isSettingsOpen, setIsSettingsOpen] = useState(false);
  
  const [isProcessing, setIsProcessing] = useState(false);
  const [progress, setProgress] = useState(0);
  const [terminalLogs, setTerminalLogs] = useState<string[]>([]);
  const logsEndRef = useRef<HTMLDivElement>(null);

  // Auto-scroll logs
  useEffect(() => {
    logsEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [terminalLogs]);

  // Load items from localforage on mount
  useEffect(() => {
    async function loadItems() {
      try {
        const saved = await localforage.getItem<RegenerationItem[]>('regenerator_items');
        if (saved && Array.isArray(saved)) {
          console.log(`[DEBUG] Loaded ${saved.length} images from local storage (localforage)`);
          setItems(saved);
        }
      } catch (e) {
        console.error("Failed to load regenerator items from localforage", e);
      } finally {
        setItemsLoaded(true);
      }
    }
    loadItems();
  }, []);

  // Save items to localforage whenever they change
  useEffect(() => {
    if (!itemsLoaded) return;
    try {
      localforage.setItem('regenerator_items', items);
    } catch (e) {
      console.error("Failed to save regenerator items to localforage", e);
    }
  }, [items, itemsLoaded]);
  
  const [aspectRatio, setAspectRatio] = useState<AspectRatioOption>('auto');
  const [imageSize, setImageSize] = useState<ImageSize>('1440p');
  const [model, setModel] = useState<ModelVersion>('gemini-3.1-flash-image');
  const [mode, setMode] = useState<'product_hero' | 'hardware_part' | 'clone'>('product_hero');
  const [customPrompt, setCustomPrompt] = useState("");
  
  const [refinePrompt, setRefinePrompt] = useState("");
  const [downloadDropdownOpen, setDownloadDropdownOpen] = useState(false);

  const isPausedRef = useRef(false);
  const isStoppedRef = useRef(false);
  const [isPaused, setIsPaused] = useState(false);

  const togglePause = () => {
    const newVal = !isPaused;
    setIsPaused(newVal);
    isPausedRef.current = newVal;
  };

  const stopBatch = () => {
    isStoppedRef.current = true;
    setIsPaused(false);
    isPausedRef.current = false;
  };

  const removeItem = (id: string) => {
    setItems((prev) => prev.filter((item) => item.id !== id));
    if (selectedItem?.id === id) {
      setSelectedItem(null);
    }
  };

  const fileInputRef = useRef<HTMLInputElement>(null);
  const folderInputRef = useRef<HTMLInputElement>(null);

  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const files = e.target.files;
    if (!files || files.length === 0) return;

    const newItems: RegenerationItem[] = [];

    for (let i = 0; i < files.length; i++) {
      const file = files[i];

      if (file.name.endsWith('.zip')) {
        const zip = new JSZip();
        const loadedZip = await zip.loadAsync(file);
        for (const [filename, zipEntry] of Object.entries(loadedZip.files)) {
          if (!zipEntry.dir && filename.match(/\.(jpg|jpeg|png|webp)$/i)) {
            const base64 = await zipEntry.async('base64');
            const ext = filename.split('.').pop()?.toLowerCase();
            const mimeType = ext === 'jpg' ? 'jpeg' : ext;
            newItems.push({
              id: Math.random().toString(36).substr(2, 9),
              filename,
              originalDataUri: `data:image/${mimeType};base64,${base64}`,
              status: 'pending',
            });
          }
        }
      } else if (file.type.startsWith('image/')) {
        const dataUri = await new Promise<string>((resolve) => {
          const reader = new FileReader();
          reader.onload = (e) => resolve(e.target?.result as string);
          reader.readAsDataURL(file);
        });
        newItems.push({
          id: Math.random().toString(36).substr(2, 9),
          filename: file.name,
          originalDataUri: dataUri,
          status: 'pending',
        });
      }
    }

    setItems((prev) => [...prev, ...newItems]);
    if (fileInputRef.current) fileInputRef.current.value = '';
    if (folderInputRef.current) folderInputRef.current.value = '';
  };

  const startBatch = async () => {
    if (items.length === 0) return;
    setIsProcessing(true);
    setIsPaused(false);
    isPausedRef.current = false;
    isStoppedRef.current = false;
    setProgress(0);

    const pendingItems = items.filter(item => item.status === 'pending' || item.status === 'error');
    
    // Concurrency allocation for different generation modes to stay well within quotas
    const CONCURRENCY_LIMIT = mode === 'clone' ? 5 : mode === 'hardware_part' ? 3 : 2;
    
    let processedCount = 0;
    
    const processItem = async (item: RegenerationItem) => {
      if (isStoppedRef.current) return;

      while (isPausedRef.current && !isStoppedRef.current) {
        await new Promise(r => setTimeout(r, 200));
      }

      if (isStoppedRef.current) return;

      setItems(prev => prev.map(p => p.id === item.id ? { ...p, status: 'processing', error: undefined } : p));

      try {
        const [mimeTypePrefix, base64Data] = item.originalDataUri.split(';base64,');
        const mimeType = mimeTypePrefix.split(':')[1];
        
        const result = await regenerateImage(
          base64Data,
          mimeType,
          customPrompt,
          aspectRatio,
          imageSize,
          model,
          mode,
          (msg) => {
            setTerminalLogs(prev => [...prev, `[${new Date().toLocaleTimeString()}] ${item.filename}: ${msg}`].slice(-200));
          }
        );

        if (isStoppedRef.current) return;

        setItems(prev => prev.map(p => p.id === item.id ? { 
          ...p, 
          status: 'success', 
          generatedDataUri: result.imageUrl 
        } : p));
      } catch (error: any) {
        console.error(`Error processing ${item.filename}:`, error);
        if (!isStoppedRef.current) {
          setItems(prev => prev.map(p => p.id === item.id ? { 
            ...p, 
            status: 'error', 
            error: error.message || 'Unknown error' 
          } : p));
        }
      } finally {
        processedCount++;
        setProgress((processedCount / pendingItems.length) * 100);
      }
    };

    const queue = [...pendingItems];
    const workers = Array(Math.min(CONCURRENCY_LIMIT, queue.length))
      .fill(null)
      .map(async (_, index) => {
        // Stagger the initial startup to prevent instantaneous rate-limit burst
        await new Promise(r => setTimeout(r, index * 600));

        while (queue.length > 0 && !isStoppedRef.current) {
          const item = queue.shift();
          if (item) {
            while (isPausedRef.current && !isStoppedRef.current) {
              await new Promise(r => setTimeout(r, 200));
            }
            if (isStoppedRef.current) {
              break;
            }
            await processItem(item);
            // Crucial spacing between items to maintain smooth API quota allocation for 100+ requests
            if (!isStoppedRef.current && queue.length > 0) {
              await new Promise(r => setTimeout(r, 1200)); 
            }
          }
        }
      });
    
    await Promise.all(workers);

    // Reset any remainder processing items to pending if we stopped midway
    if (isStoppedRef.current) {
      setItems(p => p.map(item => item.status === 'processing' ? { ...item, status: 'pending' } : item));
    }

    setIsProcessing(false);
    setIsPaused(false);
    isPausedRef.current = false;
  };

  const refineItem = async (id: string, prompt?: string) => {
    const item = items.find(p => p.id === id);
    if (!item?.generatedDataUri) return;

    setItems(prev => prev.map(p => p.id === id ? { ...p, status: 'processing', error: undefined } : p));

    try {
      const [mimeTypePrefix, base64Data] = item.generatedDataUri.split(';base64,');
      const mimeType = mimeTypePrefix.split(':')[1];
      
      const result = await refineImage(
        base64Data,
        mimeType,
        prompt || customPrompt,
        aspectRatio,
        imageSize,
        model,
        mode,
        (msg) => {
          setTerminalLogs(prev => [...prev, `[${new Date().toLocaleTimeString()}] ${item.filename} (Refine): ${msg}`].slice(-200));
        }
      );

      setItems(prev => prev.map(p => p.id === id ? { 
        ...p, 
        status: 'success', 
        generatedDataUri: result.imageUrl 
      } : p));
    } catch (error: any) {
      console.error(`Error refining ${item.filename}:`, error);
      setItems(prev => prev.map(p => p.id === id ? { 
        ...p, 
        status: 'error', 
        error: error.message || 'Unknown error' 
      } : p));
    }
  };

  const convertBase64ToFormat = async (base64DataUri: string, format: 'png' | 'webp'): Promise<Blob | null> => {
    const mimeType = base64DataUri.split(';')[0].split(':')[1];
    if (format === 'png' && mimeType === 'image/png') {
       const resp = await fetch(base64DataUri);
       return await resp.blob();
    }
    if (format === 'webp' && mimeType === 'image/webp') {
       const resp = await fetch(base64DataUri);
       return await resp.blob();
    }

    return new Promise((resolve) => {
      const img = new Image();
      img.onload = () => {
        const canvas = document.createElement("canvas");
        canvas.width = img.naturalWidth;
        canvas.height = img.naturalHeight;
        const ctx = canvas.getContext("2d");
        if (!ctx) {
          fetch(base64DataUri).then(r => r.blob()).then(resolve).catch(() => resolve(null));
          return;
        }
        ctx.drawImage(img, 0, 0);
        canvas.toBlob((resultBlob) => {
          if (resultBlob) {
            resolve(resultBlob);
          } else {
            fetch(base64DataUri).then(r => r.blob()).then(resolve).catch(() => resolve(null));
          }
        }, format === 'webp' ? 'image/webp' : 'image/png', 0.95);
      };
      img.onerror = () => {
        fetch(base64DataUri).then(r => r.blob()).then(resolve).catch(() => resolve(null));
      };
      img.src = base64DataUri;
    });
  };

  const downloadAll = async (format: 'png' | 'webp') => {
    const zip = new JSZip();
    const successfulItems = items.filter(item => item.status === 'success' && item.generatedDataUri);
    
    if (successfulItems.length === 0) return;

    for (const item of successfulItems) {
      const convertedBlob = await convertBase64ToFormat(item.generatedDataUri!, format);
      const cleanName = item.filename.replace(/\.[^/.]+$/, "") + `.${format}`;
      if (convertedBlob) {
        zip.file(cleanName, convertedBlob);
      } else {
        const base64Data = item.generatedDataUri!.split(',')[1];
        zip.file(cleanName, base64Data, { base64: true });
      }
    }

    const content = await zip.generateAsync({ type: 'blob' });
    saveAs(content, `regenerated_images_${format}.zip`);
  };

  const removeAll = () => {
    if (!showClearConfirm) {
      setShowClearConfirm(true);
      // Automatically reset the confirmation after 3 seconds of inactivity
      setTimeout(() => {
        setShowClearConfirm(false);
      }, 3000);
      return;
    }
    setItems([]);
    setShowClearConfirm(false);
  };

  return (
    <div className="fixed inset-0 z-50 bg-black/90 flex flex-col p-0 sm:p-8 overflow-y-auto overflow-x-hidden">
      <div className="max-w-7xl mx-auto w-full bg-neutral-900 sm:border border-white/10 sm:rounded-2xl flex flex-col shadow-2xl relative min-h-screen sm:min-h-[500px] h-full sm:max-h-[90vh]">
        
        {/* Header */}
        <div className="flex items-center justify-between p-4 sm:p-6 border-b border-white/5 bg-neutral-800/50 shrink-0">
          <div className="flex items-center gap-3">
            <div className="w-8 h-8 sm:w-10 sm:h-10 bg-indigo-500/20 rounded-xl flex items-center justify-center shrink-0">
              <Sparkles className="w-4 h-4 sm:w-5 sm:h-5 text-indigo-400" />
            </div>
            <div>
              <div className="flex items-center gap-2 flex-wrap">
                <h2 className="text-lg sm:text-xl font-bold text-white">Bulk Image Regenerator</h2>
                <div className="flex items-center gap-1 px-2 py-0.5 bg-emerald-500/10 border border-emerald-500/20 rounded-full text-[10px] font-medium text-emerald-400">
                  <CheckCircle2 className="w-3 h-3" />
                  <span className="hidden sm:inline">Auto-Saved to Local Storage (IndexedDB)</span>
                  <span className="sm:hidden">Auto-Saved</span>
                </div>
              </div>
              <p className="text-xs sm:text-sm text-neutral-400 hidden sm:block">Process single images or ZIP files into photorealistic renders</p>
            </div>
          </div>
          <button onClick={onClose} className="p-2 hover:bg-white/10 rounded-lg transition-colors text-neutral-400 hover:text-white shrink-0">
            <X className="w-5 h-5 sm:w-6 sm:h-6" />
          </button>
        </div>

        <div className="flex flex-col lg:flex-row flex-1 sm:overflow-hidden overflow-y-auto relative bg-neutral-900">
          {/* Sidebar Settings */}
          <div className="w-full lg:w-80 border-b lg:border-b-0 lg:border-r border-white/10 lg:overflow-y-auto bg-neutral-900/50 shrink-0 flex flex-col transition-all">
            <div 
              className="flex items-center justify-between p-4 sm:p-6 cursor-pointer lg:cursor-auto select-none"
              onClick={() => setIsSettingsOpen(!isSettingsOpen)}
            >
              <h3 className="text-lg font-semibold text-white">Settings</h3>
              <button className="lg:hidden p-1 text-neutral-400 hover:text-white transition-colors">
                {isSettingsOpen ? <ChevronUp className="w-5 h-5" /> : <ChevronDown className="w-5 h-5" />}
              </button>
            </div>
            
            <div className={`space-y-6 px-4 sm:px-6 pb-6 lg:!block ${isSettingsOpen ? 'block' : 'hidden'}`}>
              <div>
                <label className="text-sm text-neutral-400 block mb-2">Generation Mode</label>
                <div className="flex flex-col gap-2">
                  <button 
                    onClick={() => setMode('clone')}
                    disabled={isProcessing}
                    className={`px-3 py-2.5 text-left rounded-lg border flex flex-col gap-0.5 transition-colors ${mode === 'clone' ? 'bg-indigo-600/20 border-indigo-500 text-white' : 'bg-neutral-800 border-white/10 text-neutral-400 hover:bg-neutral-800/80 disabled:opacity-50'}`}
                  >
                    <span className="text-sm font-semibold text-neutral-100">1:1 Clone (Original POV)</span>
                    <span className="text-xs text-neutral-400">Matches original perspective and layout exactly</span>
                  </button>

                  <button 
                    onClick={() => setMode('product_hero')}
                    disabled={isProcessing}
                    className={`px-3 py-2.5 text-left rounded-lg border flex flex-col gap-0.5 transition-colors ${mode === 'product_hero' ? 'bg-indigo-600/20 border-indigo-500 text-white' : 'bg-neutral-800 border-white/10 text-neutral-400 hover:bg-neutral-800/80 disabled:opacity-50'}`}
                  >
                    <span className="text-sm font-semibold text-neutral-100 flex items-center gap-1.5">
                      Product Hero / Box Composition
                      <span className="px-1.5 py-0.5 text-[9px] bg-indigo-500/30 text-indigo-300 rounded font-bold uppercase tracking-wide">Dynamic</span>
                    </span>
                    <span className="text-xs text-neutral-400">Creative regrouping for tools and retail packaging</span>
                  </button>

                  <button 
                    onClick={() => setMode('hardware_part')}
                    disabled={isProcessing}
                    className={`px-3 py-2.5 text-left rounded-lg border flex flex-col gap-0.5 transition-colors ${mode === 'hardware_part' ? 'bg-indigo-600/20 border-indigo-500 text-white' : 'bg-neutral-800 border-white/10 text-neutral-400 hover:bg-neutral-800/80 disabled:opacity-50'}`}
                  >
                    <span className="text-sm font-semibold text-neutral-100 flex items-center gap-1.5">
                      Hardware Part (Precision)
                      <span className="px-1.5 py-0.5 text-[9px] bg-green-500/30 text-green-300 rounded font-bold uppercase tracking-wide">Structure Match</span>
                    </span>
                    <span className="text-xs text-neutral-400">Precise structural matching for screws and parts without creative hallucination</span>
                  </button>
                </div>
              </div>

              <div>
                <label className="text-sm text-neutral-400 block mb-2">Model</label>
                <select 
                  value={model}
                  onChange={(e) => setModel(e.target.value as ModelVersion)}
                  className="w-full bg-neutral-800 border border-white/10 rounded-lg px-3 py-2 text-sm text-white focus:outline-none focus:ring-2 focus:ring-indigo-500/50"
                  disabled={isProcessing}
                >
                  {Object.entries(MODEL_LABELS).map(([value, label]) => (
                    <option key={value} value={value}>{label}</option>
                  ))}
                </select>
              </div>

              <div className="grid grid-cols-2 lg:grid-cols-1 gap-4 lg:gap-6">
                <div>
                  <label className="text-sm text-neutral-400 block mb-2">Aspect Ratio</label>
                  <select 
                    value={aspectRatio}
                    onChange={(e) => setAspectRatio(e.target.value as AspectRatioOption)}
                    className="w-full bg-neutral-800 border border-white/10 rounded-lg px-3 py-2 text-sm text-white focus:outline-none focus:ring-2 focus:ring-indigo-500/50"
                    disabled={isProcessing}
                  >
                     <option value="auto">Auto-detect</option>
                     {Object.entries({
                          ...ASPECT_RATIO_LABELS,
                          "1:4": "1:4 (Narrow)",
                          "4:1": "4:1 (Banner)",
                          "1:8": "1:8 (Ultra-Narrow)",
                          "8:1": "8:1 (Ultra-Wide)",
                        }).map(([value, label]) => (
                          <option key={value} value={value}>{label}</option>
                        ))}
                  </select>
                </div>

                <div>
                  <label className="text-sm text-neutral-400 block mb-2">Image Size</label>
                  <select 
                    value={imageSize}
                    onChange={(e) => setImageSize(e.target.value as ImageSize)}
                    className="w-full bg-neutral-800 border border-white/10 rounded-lg px-3 py-2 text-sm text-white focus:outline-none focus:ring-2 focus:ring-indigo-500/50"
                    disabled={isProcessing}
                  >
                    {Object.entries(IMAGE_SIZE_LABELS).map(([value, label]) => (
                      <option key={value} value={value}>{label}</option>
                    ))}
                  </select>
                </div>
              </div>

              <div>
                 <label className="text-sm text-neutral-400 block mb-2">Additional Instructions (Optional)</label>
                 <textarea 
                   value={customPrompt}
                   onChange={(e) => setCustomPrompt(e.target.value)}
                   placeholder="e.g., Make it look like brushed aluminum..."
                   className="w-full bg-neutral-800 border border-white/10 rounded-lg px-3 py-2 text-sm text-white focus:outline-none focus:ring-2 focus:ring-indigo-500/50 h-24 resize-none"
                   disabled={isProcessing}
                 />
              </div>
            </div>
          </div>

          {/* Main Content */}
          <div className="flex-1 flex flex-col bg-neutral-900 border-t lg:border-t-0 lg:border-l border-white/10 lg:overflow-hidden min-h-[500px] lg:min-h-0">
            {/* Toolbar */}
            <div className="p-4 border-b border-white/10 flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4 bg-neutral-800/30 overflow-x-auto shrink-0">
              <div className="flex flex-wrap items-center gap-3">
                <input 
                  type="file" 
                  ref={fileInputRef} 
                  onChange={handleFileUpload} 
                  multiple 
                  accept="image/*,.zip" 
                  className="hidden" 
                />
                <input 
                  type="file" 
                  ref={folderInputRef} 
                  onChange={handleFileUpload} 
                  // @ts-ignore
                  webkitdirectory="true"
                  className="hidden" 
                />
                <button 
                  onClick={() => fileInputRef.current?.click()}
                  disabled={isProcessing}
                  className="px-4 py-2 bg-neutral-800 hover:bg-neutral-700 text-white rounded-lg font-medium transition-colors flex items-center gap-2 border border-white/10 disabled:opacity-50"
                >
                  <Upload className="w-4 h-4" />
                  Add Files / ZIP
                </button>
                <button 
                  onClick={() => folderInputRef.current?.click()}
                  disabled={isProcessing}
                  className="px-4 py-2 bg-neutral-800 hover:bg-neutral-700 text-white rounded-lg font-medium transition-colors flex items-center gap-2 border border-white/10 disabled:opacity-50"
                  title="Select Folder"
                >
                  <FolderOpen className="w-4 h-4" />
                  Add Folder
                </button>
                {items.length > 0 && (
                  <button 
                    onClick={removeAll}
                    disabled={isProcessing}
                    className={`px-4 py-2 rounded-lg font-medium transition-colors disabled:opacity-50 border ${
                      showClearConfirm 
                        ? 'bg-red-600/95 hover:bg-red-500 border-red-500 text-white font-semibold'
                        : 'bg-red-500/10 hover:bg-red-500/20 border-transparent text-red-400'
                    }`}
                  >
                    {showClearConfirm ? 'Click to Confirm Clear' : 'Clear All'}
                  </button>
                )}
              </div>
              
              <div className="flex flex-wrap items-center gap-3">
                {items.some(i => i.status === 'success') && (
                  <div className="relative">
                    <button 
                      onClick={() => setDownloadDropdownOpen(!downloadDropdownOpen)}
                      disabled={isProcessing && !isPaused}
                      className="px-4 py-2 bg-emerald-600 hover:bg-emerald-500 text-white rounded-lg font-medium transition-colors flex items-center gap-2 shadow-lg shadow-emerald-900/20 disabled:opacity-50"
                    >
                      <Download className="w-4 h-4" />
                      Download All ZIP <ChevronDown className="w-4 h-4" />
                    </button>
                    {downloadDropdownOpen && (
                      <div className="absolute right-0 top-full mt-1 w-48 bg-neutral-800 border border-white/10 rounded-lg shadow-xl z-50 overflow-hidden font-sans">
                        <button
                          onClick={async (e) => {
                            e.stopPropagation();
                            setDownloadDropdownOpen(false);
                            await downloadAll('png');
                          }}
                          className="w-full text-left px-4 py-2 hover:bg-indigo-600 text-white text-xs transition-colors"
                        >
                          Download as PNG (ZIP)
                        </button>
                        <button
                          onClick={async (e) => {
                            e.stopPropagation();
                            setDownloadDropdownOpen(false);
                            await downloadAll('webp');
                          }}
                          className="w-full text-left px-4 py-2 hover:bg-indigo-600 text-white text-xs transition-colors"
                        >
                          Download as WebP (ZIP)
                        </button>
                      </div>
                    )}
                  </div>
                )}
                
                {isProcessing ? (
                  <div className="flex items-center gap-2">
                    <button 
                      onClick={togglePause}
                      className={`px-4 py-2 border rounded-lg font-medium transition-colors flex items-center gap-2 shadow-sm ${
                        isPaused 
                          ? 'bg-amber-600/35 border-amber-500 text-white hover:bg-amber-600/50' 
                          : 'bg-neutral-800 border-white/10 text-neutral-200 hover:bg-neutral-700 hover:text-white'
                      }`}
                    >
                      {isPaused ? (
                        <>
                          <Play className="w-4 h-4 text-emerald-400 fill-emerald-400" />
                          Resume
                        </>
                      ) : (
                        <>
                          <Pause className="w-4 h-4 text-amber-400 fill-amber-400" />
                          Pause
                        </>
                      )}
                    </button>
                    <button 
                      onClick={stopBatch}
                      className="px-4 py-2 bg-red-600 hover:bg-red-500 text-white rounded-lg font-medium transition-colors flex items-center gap-2 shadow-lg shadow-red-900/20"
                    >
                      <Square className="w-4 h-4 text-white fill-white" />
                      Stop
                    </button>
                  </div>
                ) : (
                  <button 
                    onClick={startBatch}
                    disabled={items.filter(i => i.status === 'pending' || i.status === 'error').length === 0}
                    className="px-6 py-2 bg-indigo-600 hover:bg-indigo-500 text-white rounded-lg font-medium transition-colors flex items-center gap-2 shadow-lg shadow-indigo-900/20 disabled:opacity-50"
                  >
                    <Play className="w-4 h-4 text-indigo-200 fill-indigo-200" />
                    Start Batch ({items.filter(i => i.status === 'pending' || i.status === 'error').length})
                  </button>
                )}
              </div>
            </div>

            {/* Progress Bar */}
            {isProcessing && (
              <div className="h-1 bg-neutral-800 w-full">
                <div 
                  className="h-full bg-indigo-500 transition-all duration-300 ease-out"
                  style={{ width: `${progress}%` }}
                />
              </div>
            )}

            {/* Item List */}
            <div className="flex-1 overflow-y-auto p-6">
              {items.length === 0 ? (
                <div className="h-full flex flex-col items-center justify-center text-neutral-500">
                  <Upload className="w-12 h-12 mb-4 opacity-50 text-indigo-400" />
                  <p className="text-lg font-medium text-neutral-400">No images added</p>
                  <p className="text-sm mt-2 text-center max-w-sm text-neutral-500">
                    Upload multiple images or a ZIP file to begin generating.
                  </p>
                  <p className="text-[11px] mt-1 text-center max-w-xs text-neutral-500 bg-neutral-800/40 px-3 py-1.5 rounded-lg border border-white/5">
                    💡 All images are safely auto-stored in local storage (IndexedDB) to persist between batch failures or quota limits.
                  </p>
                </div>
              ) : (
                <div className="grid grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">
                  {items.map(item => (
                    <div key={item.id} className="bg-neutral-800 rounded-xl border border-white/5 overflow-hidden flex flex-col cursor-pointer group hover:border-indigo-500/50" onClick={() => { setSelectedItem(item); setRefinePrompt(''); }}>
                      <div className="aspect-square relative bg-neutral-900/50 border-b border-white/5">
                        <img 
                          src={item.generatedDataUri || item.originalDataUri} 
                          alt={item.filename}
                          className="w-full h-full object-contain p-2 group-hover:scale-105 transition-transform duration-200"
                        />
                        {/* Individual Remove Button */}
                        <button 
                          onClick={(e) => { 
                            e.stopPropagation(); 
                            removeItem(item.id); 
                          }}
                          disabled={isProcessing && item.status === 'processing'}
                          className="absolute top-2 left-2 bg-neutral-900/85 hover:bg-red-600 border border-white/10 text-neutral-400 hover:text-white p-1.5 rounded-lg shadow-lg opacity-0 group-hover:opacity-100 transition-all z-10 disabled:opacity-30 disabled:pointer-events-none"
                          title="Remove from batch"
                        >
                          <Trash2 className="w-3.5 h-3.5" />
                        </button>

                        {item.status === 'processing' && (
                          <div className="absolute inset-0 bg-black/50 flex items-center justify-center backdrop-blur-sm z-10">
                            <Loader2 className="w-8 h-8 text-indigo-400 animate-spin" />
                          </div>
                        )}
                        {item.status === 'success' && (
                          <div className="absolute top-2 right-2 flex gap-1 opacity-0 group-hover:opacity-100 transition-opacity z-10">
                            <button 
                              onClick={(e) => { e.stopPropagation(); refineItem(item.id); }}
                              className="bg-indigo-500 text-white p-1.5 rounded-full shadow-lg hover:bg-indigo-600 transition-colors"
                              title="Refine Image"
                            >
                              <RefreshCw className="w-4 h-4" />
                            </button>
                          </div>
                        )}
                        {item.status === 'error' && (
                          <div className="absolute top-2 right-2 bg-red-500 text-white p-1 rounded-full shadow-lg z-10" title={item.error}>
                            <AlertCircle className="w-4 h-4" />
                          </div>
                        )}
                      </div>
                      <div className="p-3 bg-neutral-800/80 max-w-full">
                        <p className="text-xs text-white truncate font-medium max-w-full" title={item.filename}>{item.filename}</p>
                        <p className="text-[10px] text-neutral-400 mt-1 capitalize max-w-full">
                          {item.status === 'error' ? <span className="text-red-400 truncate block w-full">{item.error}</span> : item.status}
                        </p>
                      </div>
                    </div>
                  ))}
                </div>
              )}
            </div>

            {/* Terminal Logs View */}
            {terminalLogs.length > 0 && (
              <div className="h-48 shrink-0 bg-[#0a0a0a] border-t border-white/10 font-mono flex flex-col z-20 shadow-[0_-10px_20px_rgba(0,0,0,0.5)]">
                <div className="flex justify-between items-center px-4 py-2 border-b border-white/5 bg-neutral-900/80">
                  <span className="text-indigo-400 font-semibold text-[11px] tracking-[0.15em] uppercase flex items-center gap-2">
                    <Sparkles className="w-3.5 h-3.5" />
                    AI Synthesis Pipeline Logs
                  </span>
                  <button onClick={() => setTerminalLogs([])} className="text-neutral-500 hover:text-red-400 p-1.5 rounded-md hover:bg-white/5 transition-colors">
                    <Trash2 className="w-3.5 h-3.5" />
                  </button>
                </div>
                <div className="flex-1 overflow-y-auto p-4 space-y-1.5 text-[11px] leading-relaxed">
                  {terminalLogs.map((log, i) => (
                    <div key={i} className={`whitespace-pre-wrap ${log.includes('WARNING') || log.includes('Validation Failed') ? 'text-amber-400' : log.includes('failed') || log.includes('error') ? 'text-red-400' : log.includes('SUCCESS') || log.includes('successfully') ? 'text-emerald-400' : 'text-neutral-300'}`}>
                      <span className="text-neutral-600 mr-2">›</span>
                      {log}
                    </div>
                  ))}
                  <div ref={logsEndRef} />
                </div>
              </div>
            )}
          </div>
        </div>
        
        {/* Review Modal */}
        {selectedItem && (
          <div className="fixed inset-0 z-[60] bg-black/90 overflow-y-auto" onClick={() => setSelectedItem(null)}>
            <div className="min-h-screen w-full flex items-start sm:items-center justify-center p-0 sm:p-4">
              <div className="max-w-4xl w-full bg-neutral-900 sm:rounded-2xl flex flex-col p-4 sm:p-6 shadow-2xl relative min-h-screen sm:min-h-0" onClick={(e) => e.stopPropagation()}>
                <div className="flex justify-between items-center mb-6 shrink-0">
                  <h3 className="text-lg sm:text-xl font-bold text-white truncate pr-4">{selectedItem.filename}</h3>
                  <button onClick={() => setSelectedItem(null)} className="text-neutral-400 hover:text-white shrink-0 bg-neutral-800/50 hover:bg-neutral-800 p-2 rounded-lg transition-colors">
                    <X className="w-5 h-5 sm:w-6 sm:h-6" />
                  </button>
                </div>
                
                <div className={`relative w-full bg-neutral-900 border border-white/10 rounded-xl overflow-hidden grid ${selectedItem.generatedDataUri ? 'grid-cols-1 lg:grid-cols-2 gap-px bg-white/10' : 'grid-cols-1'}`}>
                  <div className="relative w-full flex flex-col bg-black">
                    {selectedItem.generatedDataUri && (
                      <div className="absolute top-3 left-3 z-10 bg-black/60 backdrop-blur-md px-2.5 py-1.5 rounded-lg text-xs font-semibold tracking-wide text-white border border-white/10 shadow-lg">Original</div>
                    )}
                    <img 
                      src={selectedItem.originalDataUri} 
                      alt={`Original ${selectedItem.filename}`}
                      className="w-full h-auto object-contain p-4 bg-dots"
                    />
                  </div>
                  {selectedItem.generatedDataUri && (
                    <div className="relative w-full flex flex-col bg-black">
                      <div className="absolute top-3 left-3 z-10 bg-indigo-500/80 backdrop-blur-md px-2.5 py-1.5 rounded-lg text-xs font-semibold tracking-wide text-white border border-indigo-500/50 shadow-lg shadow-indigo-500/20">Regenerated</div>
                      <img 
                        src={selectedItem.generatedDataUri} 
                        alt={`Regenerated ${selectedItem.filename}`}
                        className="w-full h-auto object-contain p-4 bg-dots"
                      />
                    </div>
                  )}
                </div>
                
                {selectedItem.status === 'success' && (
                  <div className="mt-6 flex flex-col sm:flex-row gap-3 shrink-0">
                    <input 
                      type="text" 
                      value={refinePrompt} 
                      onChange={(e) => setRefinePrompt(e.target.value)}
                      placeholder="Enter refinement instructions to fix this generated image..."
                      className="flex-1 bg-neutral-800 border border-white/10 rounded-xl px-4 py-3 sm:py-3 text-sm sm:text-base text-white focus:outline-none focus:ring-2 focus:ring-indigo-500/50 placeholder:text-neutral-500"
                      onKeyDown={(e) => {
                        if (e.key === 'Enter' && refinePrompt.trim()) {
                          refineItem(selectedItem.id, refinePrompt.trim());
                          setSelectedItem(null);
                          setRefinePrompt('');
                        }
                      }}
                    />
                    <button 
                      onClick={() => {
                        if (refinePrompt.trim()) {
                          refineItem(selectedItem.id, refinePrompt.trim());
                          setSelectedItem(null);
                          setRefinePrompt('');
                        }
                      }}
                      disabled={!refinePrompt.trim()}
                      className="bg-indigo-600 hover:bg-indigo-500 disabled:opacity-50 disabled:hover:bg-indigo-600 text-white px-8 py-3 rounded-xl font-medium w-full sm:w-auto transition-colors flex items-center justify-center gap-2"
                    >
                      <Sparkles className="w-4 h-4" />
                      Refine
                    </button>
                  </div>
                )}
              </div>
            </div>
          </div>
        )}
      </div>
    </div>
  );
}
